def SignalMessage.tryFromL181–199 ↗#

SignalMessage::try_from(&[u8]) (protocol.rs:261–307): length ≥ 9, version 3 ≤ value[0] >> 4 ≤ 4, protobuf decode of the bytes between the version byte and the MAC, required ratchet_key (then PublicKey::deserialize), counter and ciphertext; an absent previous_counter is 0 and an absent pq_ratchet is empty.

def SignalMessage.tryFrom (codec : SignalMessageProtoCodec) (value : List (BitVec 8)) : Except SignalProtocolError SignalMessage := do if value.length < macLength + 1 then throw (.ciphertextMessageTooShort value.length) let v := (value.headD 0) >>> 4 if v < preKyberVersion then throw (.legacyCiphertextVersion v) if currentVersion < v then throw (.unrecognizedCiphertextVersion v) let some p := codec.decode ((value.take (value.length - macLength)).drop 1) | throw .invalidProtobufEncoding let some rk := p.ratchetKey | throw .invalidProtobufEncoding let senderRatchetKey ← PublicKey.deserialize rk let some counter := p.counter | throw .invalidProtobufEncoding let some ciphertext := p.ciphertext | throw .invalidProtobufEncoding return { messageVersion := v, senderRatchetKey, counter, previousCounter := p.previousCounter.getD 0, ciphertext, pqRatchet := p.pqRatchet.getD [], addresses := p.addresses, serialized := value }
